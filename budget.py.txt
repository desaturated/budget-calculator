def get_positive_float(prompt):
    while True:
        try:
            value = float(input(prompt))
            if value < 0:
                print("Please enter a positive number.")
                continue
            return value
        except ValueError:
            print("Invalid input. Please enter a number.")

print("Welcome to the Budgeting Program!")

budget = get_positive_float("What is your total budget? $")

categories = ["Housing", "Food", "Clothes", "Hobbies"]
spending = {}

total_spent = 0

for category in categories:
    amount = get_positive_float(f"How much do you spend on {category}? $")
    spending[category] = amount
    total_spent += amount

print("\n--- Budget Summary ---")
for category, amount in spending.items():
    print(f"{category}: ${amount:.2f}")

print(f"\nTotal spending: ${total_spent:.2f}")

if total_spent > budget:
    over = total_spent - budget
    print(f"You went over your budget by ${over:.2f}")
else:
    left = budget - total_spent
    print(f"You stayed within your budget! You have ${left:.2f} left.")
